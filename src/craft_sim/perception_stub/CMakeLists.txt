# 시뮬레이터 전용 craft_perception 대체 빌드: ncnn·OpenCV 없이 플래닝이 쓰는 메시지와 tag_registry만 만든다.
# 원본 소스(msg, include, src)는 CRAFT_SRC/craft_perception 를 그대로 쓴다.
cmake_minimum_required(VERSION 3.14)
project(craft_perception)
if(NOT DEFINED CRAFT_SRC)
  set(CRAFT_SRC "$ENV{CRAFT_SRC}")
endif()
set(P "${CRAFT_SRC}/craft_perception")
if(NOT EXISTS "${P}/msg")
  message(FATAL_ERROR "CRAFT_SRC/craft_perception 를 찾을 수 없음: ${P}")
endif()
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
file(GLOB MSGS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/msg/*.msg")
rosidl_generate_interfaces(${PROJECT_NAME} ${MSGS} DEPENDENCIES std_msgs geometry_msgs)
add_library(perception_modules "${P}/src/tag_registry.cpp")
target_compile_features(perception_modules PUBLIC cxx_std_17)
target_include_directories(perception_modules PUBLIC
  $<BUILD_INTERFACE:${P}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(perception_modules PUBLIC yaml-cpp)
install(TARGETS perception_modules EXPORT perception_modules_export
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY "${P}/include/" DESTINATION include)
install(DIRECTORY "${P}/config" DESTINATION share/${PROJECT_NAME})
ament_export_targets(perception_modules_export HAS_LIBRARY_TARGET)
ament_export_dependencies(yaml-cpp rosidl_default_runtime)
ament_package()
