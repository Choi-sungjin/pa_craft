cmake_minimum_required(VERSION 3.16)
project(driving_sim CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(craft_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(yaml-cpp REQUIRED)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../src)
add_executable(drive_plant drive_plant.cpp ${SRC}/craft_control/src/lateral_controller.cpp ${SRC}/craft_control/src/longitudinal_controller.cpp)
target_include_directories(drive_plant PRIVATE ${SRC}/craft_control/include)
ament_target_dependencies(drive_plant rclcpp craft_interfaces nav_msgs geometry_msgs sensor_msgs std_msgs tf2_ros)
target_link_libraries(drive_plant yaml-cpp)
