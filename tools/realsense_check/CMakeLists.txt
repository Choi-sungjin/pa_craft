cmake_minimum_required(VERSION 3.16)
project(realsense_check CXX)
set(CMAKE_BUILD_TYPE Release)
find_package(realsense2 REQUIRED)
add_executable(rs_snap rs_snap.cpp)
target_compile_features(rs_snap PRIVATE cxx_std_17)
target_link_libraries(rs_snap realsense2::realsense2)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui)
add_executable(rs_live rs_live.cpp)
target_compile_features(rs_live PRIVATE cxx_std_17)
target_link_libraries(rs_live realsense2::realsense2 ${OpenCV_LIBS})
add_executable(rs_depth rs_depth.cpp)
target_compile_features(rs_depth PRIVATE cxx_std_17)
target_link_libraries(rs_depth realsense2::realsense2 ${OpenCV_LIBS})
add_executable(rs_stream rs_stream.cpp)
target_compile_features(rs_stream PRIVATE cxx_std_17)
target_link_libraries(rs_stream realsense2::realsense2)
